# Web Search 并行广搜 + 可选深搜：最终执行规范

> 本文档经过三轮评审迭代，已收口所有已知问题。可直接作为 Codex 执行规范落地。

---

## 一、设计依据

**Tavily**：官方文档建议 `include_answer`、`include_raw_content`、`max_results` 必须手动设置，`search_depth=advanced` 消耗更多 credits。多 query 场景必须显式控制 Tavily 调用次数。

**SearXNG**：Search API 要求 `format=json`，服务端 `settings.yml` 必须启用对应格式，否则可能返回 403。工具层应保留错误上下文日志，避免多 query 并发后排障困难。

**Python asyncio**：`asyncio.Semaphore` 适合限制并发协程数；`asyncio.gather(..., return_exceptions=True)` 让单个 query 失败不拖垮整体。

**DDGS**：提供 `DDGS().text()`、`images()` 等能力，适合作为免费缓冲搜索源，不承担强 SLA 主搜索职责。

---

## 二、最终目标

```
query                              → precise search → 单 query 精搜
queries                            → broad search  → 多 query 并行广搜
queries + fetch_top_pages=True     → deep search   → 多 query 广搜 + Top 页面正文
```

保留一个工具：`WebSearchTool`
保留一个调度器：`SearchCoordinator`

不新增：`BroadSearchCoordinator` / `DeepSearchCoordinator` / `Reranker` / `SourceQualityScorer`

## 六、新增：`chat/application/web_search/utils/params.py`

> 公共参数规范化工具，两个文件共用，避免重复实现。
> 函数名不加下划线（公共工具模块）。不使用 `__all__`。

```python
from __future__ import annotations

from typing import Any


def normalize_bool(value: Any) -> bool:
    """将任意值规范化为 bool。

    字符串仅识别明确的 truthy 值（true / 1 / yes / y）和明确的 falsy 值
    （false / 0 / no / n），其余字符串一律视为 False。
    非字符串走标准 bool() 转换。
    """
    if isinstance(value, bool):
        return value

    if isinstance(value, str):
        v = value.strip().lower()
        if v in {"true", "1", "yes", "y"}:
            return True
        if v in {"false", "0", "no", "n"}:
            return False
        return False

    return bool(value)


def normalize_int(
    value: Any,
    *,
    default: int,
    minimum: int,
    maximum: int,
) -> int:
    """将任意值规范化为有界整数。无法转换时使用 default，然后 clamp 到 [minimum, maximum]。"""
    try:
        number = int(value)
    except (TypeError, ValueError):
        number = default

    return max(minimum, min(number, maximum))
```

放在 `search()` 方法之后：

```python
async def search_many(
    self,
    queries: List[str],
    *,
    max_results_per_query: int = 5,
    final_max_results: int = DEFAULT_FINAL_RESULTS,
    with_images: bool = False,
    freshness_required: bool = False,
    allow_paid_fallback: bool = False,
    concurrency: int = MAX_BROAD_SEARCH_CONCURRENCY,
    dedupe_domains: bool = DEFAULT_DEDUPE_DOMAINS,
    max_per_domain: int = DEFAULT_MAX_PER_DOMAIN,
) -> SearchResponse:
    normalized_queries = _normalize_queries(
        queries,
        limit=MAX_BROAD_SEARCH_QUERIES,
    )

    if not normalized_queries:
        return SearchResponse(query="", results=(), images=(), source="multi")

    max_results_per_query = normalize_int(
        max_results_per_query,
        default=5,
        minimum=1,
        maximum=MAX_RESULTS_PER_QUERY,
    )
    final_max_results = normalize_int(
        final_max_results,
        default=DEFAULT_FINAL_RESULTS,
        minimum=1,
        maximum=MAX_FINAL_RESULTS,
    )
    concurrency = normalize_int(
        concurrency,
        default=MAX_BROAD_SEARCH_CONCURRENCY,
        minimum=1,
        maximum=MAX_BROAD_SEARCH_CONCURRENCY,
    )
    max_per_domain = normalize_int(
        max_per_domain,
        default=DEFAULT_MAX_PER_DOMAIN,
        minimum=1,
        maximum=MAX_PER_DOMAIN,
    )

    semaphore = asyncio.Semaphore(concurrency)

    async def _run_one(query: str) -> Optional[SearchResponse]:
        async with semaphore:
            return await self.search(
                query=query,
                max_results=max_results_per_query,
                with_images=with_images,
                freshness_required=freshness_required,
                allow_paid_fallback=False,  # 多 query 内部搜索绕过 Tavily
            )

    tasks = [_run_one(query) for query in normalized_queries]
    raw_responses = await asyncio.gather(*tasks, return_exceptions=True)

    responses: List[SearchResponse] = []
    failures: List[str] = []

    for query, result in zip(normalized_queries, raw_responses):
        if isinstance(result, Exception):
            failures.append(f"{query}: {type(result).__name__}: {result}")
            continue
        if result is None:
            failures.append(f"{query}: returned_none")
            continue
        if not _has_content(result):
            failures.append(f"{query}: empty_result")
            continue
        responses.append(result)

    merged = _merge_many_search_responses(
        query=" | ".join(normalized_queries),
        responses=responses,
        final_max_results=final_max_results,
        dedupe_domains=dedupe_domains,
        max_per_domain=max_per_domain,
    )

    tavily_used = False

    if (
        allow_paid_fallback
        and len(merged.results) < PAID_FALLBACK_MIN_RESULTS
        and normalized_queries
    ):
        try:
            paid_response = await self._search_tavily(
                query=normalized_queries[0],
                max_results=max_results_per_query,
                with_images=with_images,
            )
            tavily_used = True
        except Exception as exc:
            failures.append(f"tavily_paid_once: {type(exc).__name__}: {exc}")
            paid_response = None

        # 注意：paid fallback 结果仅参与当次合并，不写入普通 query 缓存，
        # 避免将补充性结果污染 normalized_queries[0] 的独立搜索缓存。
        if paid_response is not None and _has_content(paid_response):
            paid_response = _with_source(paid_response, "tavily")
            merged = _merge_many_search_responses(
                query=" | ".join(normalized_queries),
                responses=[merged, paid_response],
                final_max_results=final_max_results,
                dedupe_domains=dedupe_domains,
                max_per_domain=max_per_domain,
            )

    log_ok(
        "联网广搜",
        queries=len(normalized_queries),
        results=len(merged.results),
        images=len(merged.images),
        allow_paid_fallback=allow_paid_fallback,
        tavily_used=tavily_used,
        paid_fallback_min_results=PAID_FALLBACK_MIN_RESULTS,
        failures=failures,
    )

    return merged
```

### 7.5 模块级 helper（放在文件底部，全部下划线前缀）

```python
def _normalize_queries(
    queries: List[str],
    *,
    limit: int,
) -> List[str]:
    """规范化 queries：去空、折叠空白、大小写去重、数量截断。"""
    normalized: List[str] = []
    seen: Set[str] = set()

    for query in queries:
        if not isinstance(query, str):
            continue
        value = " ".join(query.strip().split())
        if not value:
            continue
        key = value.lower()
        if key in seen:
            continue
        seen.add(key)
        normalized.append(value)
        if len(normalized) >= limit:
            break

    return normalized


def _deduplicate_results_by_url(
    results: Tuple[SearchResult, ...],
) -> Tuple[SearchResult, ...]:
    seen: Set[str] = set()
    deduped: List[SearchResult] = []

    for result in results:
        key = result.url.strip()
        if not key or key in seen:
            continue
        seen.add(key)
        deduped.append(result)

    return tuple(deduped)


def _deduplicate_images(
    images: Tuple[ImageResult, ...],
) -> Tuple[ImageResult, ...]:
    seen: Set[str] = set()
    deduped: List[ImageResult] = []

    for image in images:
        key = image.url.strip()
        if not key or key in seen:
            continue
        seen.add(key)
        deduped.append(image)

    return tuple(deduped)


def _merge_many_search_responses(
    *,
    query: str,
    responses: List[SearchResponse],
    final_max_results: int,
    dedupe_domains: bool,
    max_per_domain: int,
) -> SearchResponse:
    results: List[SearchResult] = []
    images: List[ImageResult] = []
    sources: List[str] = []

    for response in responses:
        results.extend(response.results)
        images.extend(response.images)
        if response.source:
            sources.append(response.source)

    deduped_results = _deduplicate_results_by_url(tuple(results))

    if dedupe_domains:
        deduped_results = deduplicate_results_by_domain(
            deduped_results,
            max_per_domain=max_per_domain,
        )

    deduped_images = _deduplicate_images(tuple(images))

    source = "multi"
    if sources:
        source = "multi:" + ",".join(sorted(set(sources)))

    return SearchResponse(
        query=query,
        results=deduped_results[:final_max_results],
        images=deduped_images[:final_max_results],
        source=source,
    )


def _has_content(response: SearchResponse) -> bool:
    return bool(response.answer or response.results or response.images)


def _with_source(response: SearchResponse, source: str) -> SearchResponse:
    return SearchResponse(
        query=response.query,
        results=response.results,
        answer=response.answer,
        images=response.images,
        source=source,
    )
```

### 8.5 `execute()` 核心逻辑

```python
async def execute(self, context: Dict[str, Any], **kwargs: Any) -> str:
    session_id: Optional[str] = context.get("session_id")
    if not session_id:
        return "[Tool Error] Missing session_id in execution context."

    queries = _get_queries(kwargs)
    if not queries:
        return "[Tool Error] Missing required query or queries parameter."

    max_results = normalize_int(kwargs.get("max_results"), default=5, minimum=1, maximum=10)
    final_max_results = normalize_int(kwargs.get("final_max_results"), default=12, minimum=1, maximum=20)
    with_images = normalize_bool(kwargs.get("with_images", False))
    freshness_required = normalize_bool(kwargs.get("freshness_required", False))
    fetch_top_pages = normalize_bool(kwargs.get("fetch_top_pages", False))
    fetch_top_pages_limit = normalize_int(kwargs.get("fetch_top_pages_limit"), default=2, minimum=1, maximum=3)
    fetched_page_max_chars = normalize_int(kwargs.get("fetched_page_max_chars"), default=3000, minimum=500, maximum=6000)
    allow_paid_fallback = normalize_bool(kwargs.get("allow_paid_fallback", False))
    dedupe_domains = normalize_bool(kwargs.get("dedupe_domains", True))
    max_per_domain = normalize_int(kwargs.get("max_per_domain"), default=2, minimum=1, maximum=5)

    try:
        if len(queries) == 1:
            mode = "precise"
            response = await self._coordinator.search(
                query=queries[0],
                max_results=max_results,
                with_images=with_images,
                freshness_required=freshness_required,
                allow_paid_fallback=True,
            )
        else:
            mode = "broad"
            response = await self._coordinator.search_many(
                queries=queries,
                max_results_per_query=max_results,
                final_max_results=final_max_results,
                with_images=with_images,
                freshness_required=freshness_required,
                allow_paid_fallback=allow_paid_fallback,
                concurrency=3,
                dedupe_domains=dedupe_domains,
                max_per_domain=max_per_domain,
            )

    except Exception as exc:
        log_fail(
            "联网搜索工具",
            exc,
            session_id=session_id,
            queries=queries,
            max_results=max_results,
            with_images=with_images,
            freshness_required=freshness_required,
        )
        return "[Tool Error] Unexpected error while searching the web."

    if response is None:
        return "[Tool Result] Failed to search the web."

    if not _has_search_content(response):
        return "[Tool Result] No results found for the query."

    extra_contents: List[str] = []

    if fetch_top_pages:
        mode = "deep"
        extra_contents = await self._fetch_top_pages(
            response,
            limit=fetch_top_pages_limit,
            max_chars_per_page=fetched_page_max_chars,
        )

    return _format_response(
        response,
        mode=mode,
        queries=queries,
        extra_contents=extra_contents,
    )
```

### 8.6 `_fetch_top_pages`

```python
async def _fetch_top_pages(
    self,
    response: SearchResponse,
    *,
    limit: int,
    max_chars_per_page: int,
) -> List[str]:
    """抓取合并结果的前 N 个页面正文。

    FetchCoordinator.fetch() 返回 str | FetchedDocument | None。
    非字符串结果（如 PDF 等文档类型）跳过并记录日志，不拼接正文。
    """
    if self._fetcher is None:
        return []

    contents: List[str] = []

    for result in response.results[:limit]:
        url = result.url.strip()
        if not url:
            continue

        try:
            content = await self._fetcher.fetch(url)
        except Exception as exc:
            log_fail("搜索结果页面抓取失败", exc, url=url)
            continue

        if content is None:
            continue

        if not isinstance(content, str):
            log_fail(
                "搜索结果页面跳过",
                "抓取结果不是文本 Markdown，跳过正文拼接",
                url=url,
                content_type=type(content).__name__,
            )
            continue

        content = content.strip()
        if not content:
            continue

        contents.append(
            f"Fetched page: {url}\n"
            f"{content[:max_chars_per_page]}"
        )

    return contents
```

```python
def _get_queries(kwargs: Dict[str, Any]) -> List[str]:
    raw_queries = kwargs.get("queries")
    if isinstance(raw_queries, list):
        queries = [
            item for item in raw_queries
            if isinstance(item, str) and item.strip()
        ]
        if queries:
            return queries

    query = kwargs.get("query")
    if isinstance(query, str) and query.strip():
        return [query]

    return []


def _has_search_content(response: SearchResponse) -> bool:
    return bool(response.answer or response.results or response.images)


def _format_response(
    response: SearchResponse,
    *,
    mode: str,
    queries: List[str],
    extra_contents: Optional[List[str]] = None,
) -> str:
    unique_domains = count_unique_domains(tuple(response.results))

    lines = ["[Tool Result] Web search results"]
    lines.append(f"Mode: {mode}")

    if queries:
        lines.append("Queries:")
        for query in queries:
            lines.append(f"- {query}")

    if response.source:
        lines.append(f"Source: {response.source}")

    lines.append(
        f"Summary: {len(response.results)} results, "
        f"{len(response.images)} query-level images, "
        f"{unique_domains} unique domains."
    )

    if response.source and "stale_cache" in response.source:
        lines.append(
            "Note: Some results came from stale cache and may be outdated."
        )

    if response.answer:
        lines.append(f"\nAnswer:\n{response.answer}")

    if response.results:
        lines.append("\nResults:")

    for index, result in enumerate(response.results, 1):
        title = result.title.strip() or result.url or "(no title)"
        url = result.url.strip()
        snippet = result.snippet.strip()
        domain = extract_domain(url)

        lines.append(f"\n{index}. {title}")
        if domain:
            lines.append(f"   Domain: {domain}")
        if url:
            lines.append(f"   URL: {url}")
        if snippet:
            lines.append(f"   Snippet: {snippet}")
        if result.images:
            lines.append("   Images:")
            for image in result.images[:2]:
                lines.append(_format_image_line(image, indent="      "))

    if response.images:
        lines.append("\nQuery-level images:")
        for image in response.images[:5]:
            lines.append(_format_image_line(image, indent="   "))

    if extra_contents:
        lines.append("\nFetched top pages:")
        for index, content in enumerate(extra_contents, 1):
            lines.append(f"\n--- Page {index} ---")
            lines.append(content)

    return _normalize_tool_result("\n".join(lines))


def _format_image_line(image: ImageResult, *, indent: str) -> str:
    url = image.url.strip()
    desc = image.desc.strip() if image.desc else ""
    line = f"{indent}- {url}"

    details: List[str] = []
    if desc:
        details.append(desc)
    if image.resolution:
        details.append(f"resolution={image.resolution}")
    if image.source_url:
        details.append(f"source={image.source_url}")

    if details:
        line += f" ({'; '.join(details)})"

    return line


def _normalize_tool_result(result: str) -> str:
    result = result.strip()
    if len(result) > settings.TOOL_RESULT_MAX_CHARS:
        limit = settings.TOOL_RESULT_MAX_CHARS
        keep_len = max(0, limit - len(_TRUNCATION_MARKER))
        result = result[:keep_len].rstrip() + _TRUNCATION_MARKER
    return result
```
