# Web Fetch 与 Document Parse 彻底解耦执行文档

## 0. 最终目标

当前仓库里，`web_fetch` 已经混入文档解析、OCR、Docling、PDF 保护、文档 URL 判断、`force_ocr`、`force_browser` 等职责。现有 `WebFetchTool` 的描述和 schema 明确支持文档直链、`force_browser`、`force_ocr`，并且包含 OCR supplement、auto OCR、OCR-only 等路径。

本任务目标是：

```text
Web Fetch：
只负责 URL 请求、Content-Type 判断、HTML 正文提取、文件直链下载。

Document Parse：
只负责二进制文档解析，接受文件路径或二进制流，按后缀路由解析链路。

两者职责独立，互不调用。
```

设计依据：已确定的架构文档明确规定，Web Fetch 负责网络请求、Content-Type、HTML 正文提取、文件直链下载；不负责文档内容解析和文件格式识别。文档解析工具接受文件路径或二进制流，并按后缀路由。

```python
import asyncio
from typing import Any, Dict, Optional

from chat.application.tool_content_store import cache_and_window, format_windowed_content
from chat.application.web_fetch.fetch_coordinator import FetchCoordinator
from chat.application.web_fetch.utils.url import UrlSecurityError, validate_public_http_url
from chat.core.config.app_settings import settings
from chat.domain.interfaces.tool import BaseTool
from common.logger import log_fail, log_ok


TOOL_DESCRIPTION = (
    "Fetches and extracts textual content from a web URL, returning clean Markdown. "
    "For HTML pages, this tool fetches the page and extracts readable content. "
    "For direct file links, this tool should only fetch/download the binary through the existing file pipeline; "
    "document content parsing belongs to document_parse."
)

TOOL_SCHEMA = {
    "type": "object",
    "properties": {
        "url": {
            "type": "string",
            "description": "The URL to fetch. Must start with http:// or https://.",
        },
    },
    "required": ["url"],
}


class WebFetchTool(BaseTool):
    def __init__(self, fetcher: FetchCoordinator):
        self._fetcher = fetcher

    @property
    def name(self) -> str:
        return "web_fetch"

    @property
    def description(self) -> str:
        return TOOL_DESCRIPTION

    @property
    def parameters_schema(self) -> Dict[str, Any]:
        return TOOL_SCHEMA

    async def execute(self, context: Dict[str, Any], **kwargs) -> str:
        session_id: Optional[str] = context.get("session_id")
        if not session_id:
            return "[Tool Error] Missing session_id in execution context."

        url: str = str(kwargs.get("url", "")).strip()
        if not url:
            return "[Tool Error] Missing required url parameter"

        try:
            await asyncio.to_thread(validate_public_http_url, url)
        except UrlSecurityError as e:
            return f"[Tool Error] URL rejected by security policy: {e}"

        try:
            markdown = await self._fetcher.fetch(url)
        except UrlSecurityError as e:
            return f"[Tool Error] URL rejected by security policy: {e}"
        except Exception as e:
            log_fail("网页抓取工具", e, session_id=session_id, url=url)
            return "[Tool Error] Unexpected error while fetching web page content."

        if markdown is None:
            return "[Tool Result] Failed to fetch web page content (all fetch methods exhausted)"

        markdown = markdown.strip()
        if not markdown:
            return "[Tool Result] Failed to fetch web page content (empty content returned)"

        return self._cache_and_format(
            session_id=session_id,
            url=url,
            text=markdown,
        )

    def _cache_and_format(
        self,
        *,
        session_id: str,
        url: str,
        text: str,
    ) -> str:
        metadata: Dict[str, Any] = {
            "content_kind": "web_page",
        }

        window = cache_and_window(
            session_id=session_id,
            tool_name=self.name,
            source=url,
            text=text,
            content_type="text/markdown",
            metadata=metadata,
            offset=0,
            limit=settings.TOOL_RESULT_MAX_CHARS,
        )

        return format_windowed_content(window)
```

### 目标修改

```python
from typing import Optional, List

from cachetools import TTLCache

from chat.application.web_fetch.fetcher import StaticFetcher, SteelFetcher, LocalScriptFetcher
from chat.application.web_fetch.content_processor import ContentProcessor
from chat.application.web_fetch.utils.url import validate_public_http_url, UrlSecurityError
from common.logger import log_ok, log_fail


class FetchCoordinator:
    def __init__(
        self,
        static_fetcher: StaticFetcher,
        steel_fetcher: SteelFetcher,
        local_script_fetcher: LocalScriptFetcher,
        processor: ContentProcessor,
        min_content_length: int,
        last_resort_min_length: int,
        cache_ttl_seconds: int,
        cache_max_items: int,
    ):
        self._min_content_length = min_content_length
        self._last_resort_min_length = last_resort_min_length
        self._cache: TTLCache[str, str] = TTLCache(
            maxsize=cache_max_items,
            ttl=cache_ttl_seconds,
        )

        self._static_fetcher = static_fetcher
        self._steel_fetcher = steel_fetcher
        self._local_script_fetcher = local_script_fetcher
        self._processor = processor

        self._chain = [
            (self._static_fetcher, "raw"),
            (self._steel_fetcher, "markdown"),
            (self._local_script_fetcher, "markdown"),
        ]

    async def fetch(self, url: str) -> Optional[str]:
        url = url.strip()

        try:
            validate_public_http_url(url)
        except UrlSecurityError:
            log_fail("安全校验", "URL 被拒绝", url=url)
            raise

        cached = self._get_cached(url)
        if cached is not None:
            log_ok("网页抓取缓存命中", url=url, length=len(cached))
            return cached

        failure_reasons: List[str] = []

        for index, (fetcher, content_type) in enumerate(self._chain):
            fetcher_name = fetcher.__class__.__name__
            is_last = index == len(self._chain) - 1
            min_length = self._last_resort_min_length if is_last else self._min_content_length

            try:
                content = await fetcher.fetch(url)
            except UrlSecurityError:
                raise
            except Exception as e:
                failure_reasons.append(f"{fetcher_name}: 异常={e.__class__.__name__}")
                log_fail("网页抓取", e, url=url, fetcher=fetcher_name)
                continue

            if not content:
                failure_reasons.append(f"{fetcher_name}: 内容为空")
                log_fail("网页抓取", "抓取内容为空", url=url, fetcher=fetcher_name)
                continue

            if content_type == "markdown":
                result = str(content).strip()
                if len(result) < min_length:
                    failure_reasons.append(f"{fetcher_name}: 内容过短({len(result)}字符，阈值{min_length})")
                    log_fail("网页抓取", "内容过短，触发降级", url=url, fetcher=fetcher_name)
                    continue
            else:
                result = await self._processor.process_async(content)
                if result is None:
                    failure_reasons.append(f"{fetcher_name}: 内容处理失败")
                    log_fail("网页抓取", "内容处理失败，触发降级", url=url, fetcher=fetcher_name)
                    continue

            log_ok("网页抓取", url=url, fetcher=fetcher_name, length=len(result))
            self._set_cached(url, result)
            return result

        log_fail(
            "网页抓取",
            "所有抓取器均失败",
            url=url,
            reasons=" | ".join(failure_reasons[-5:]),
        )
        return None

    def _get_cached(self, url: str) -> Optional[str]:
        self._cache.expire()
        return self._cache.get(url)

    def _set_cached(self, url: str, markdown: str) -> None:
        self._cache.expire()
        self._cache[url] = markdown
```

## 2.3 `ContentProcessor`

当前 `ContentProcessor` 在处理 `bytes` 时直接调用 `DocumentParser`，这是 Web Fetch 与文档解析耦合的核心。用户贴出的代码中，`ContentProcessor` 初始化时构造 `DocumentParser`，`process()` / `process_async()` 遇到 `bytes` 就进入文档解析。

必须改为：

```text
ContentProcessor 只处理 str。
bytes 不在这里解析。
如果 StaticFetcher 返回 bytes，ContentProcessor 返回 None 或抛出明确错误，由上层文件直链逻辑接管。
```

### 目标 `ContentProcessor`

```python
from typing import Optional

import trafilatura

from chat.application.web_fetch.utils.page import PageBlockDetection, detect_page_block, should_degrade_detection
from chat.application.web_fetch.utils.text import normalize_text
from common.logger import log_event, log_ok, log_fail


HTML_DETECTION_SCAN_CHARS = 1024


def looks_like_html(text: str) -> bool:
    lower_head = text[:HTML_DETECTION_SCAN_CHARS].lower()
    return "<html" in lower_head or "<!doctype html" in lower_head or "<body" in lower_head


def extract_markdown_from_html(html: str) -> Optional[str]:
    try:
        extracted = trafilatura.extract(
            html,
            output_format="markdown",
            include_comments=False,
            include_tables=True,
            include_links=True,
            favor_precision=False,
        )
    except Exception as e:
        log_fail("HTML 清洗", e, extractor="trafilatura")
        return None

    if not extracted:
        log_fail("HTML 清洗", "Trafilatura 未提取到正文，触发降级")
        return None

    return normalize_text(extracted)


def log_page_block(stage: str, detection: PageBlockDetection) -> None:
    log_event(
        stage,
        reason="疑似反爬/登录/JS 空壳页面",
        kind=detection.kind,
        confidence=detection.confidence,
        score=detection.score,
        signals=",".join(detection.signals[:5]),
    )


def should_reject_page(text: str, *, stage: str, html: str = "") -> bool:
    detection = detect_page_block(text, html=html)

    if not should_degrade_detection(detection):
        return False

    log_page_block(stage, detection)
    return True


class ContentProcessor:
    def __init__(self, min_content_length: int = 400):
        self._min_content_length = min_content_length

    def process(self, content: str | bytes) -> Optional[str]:
        if isinstance(content, bytes):
            log_fail("内容处理", "收到 bytes 内容，Web Fetch 不再解析文档")
            return None

        return self._process_text(content)

    async def process_async(self, content: str | bytes) -> Optional[str]:
        return self.process(content)

    def _process_text(self, content: str) -> Optional[str]:
        stripped = content.strip()
        if not stripped:
            return None

        if looks_like_html(stripped):
            if should_reject_page(stripped, stage="内容检测", html=stripped):
                return None

            return self._process_html(stripped)

        if should_reject_page(stripped, stage="内容检测"):
            return None

        return self._process_plain_text(stripped)

    def _process_html(self, html: str) -> Optional[str]:
        result = extract_markdown_from_html(html)
        if result is None:
            return None

        if should_reject_page(result, stage="HTML 清洗", html=html):
            return None

        if len(result) < self._min_content_length:
            log_fail(
                "HTML 清洗",
                f"清洗后文本过短({len(result)}字符)，阈值{self._min_content_length}，触发降级",
            )
            return None

        log_ok("HTML 清洗", extractor="trafilatura", length=len(result))
        return result

    def _process_plain_text(self, text: str) -> Optional[str]:
        normalized = normalize_text(text)

        if len(normalized) < self._min_content_length:
            log_fail(
                "纯文本检测",
                f"文本过短({len(normalized)}字符)，阈值{self._min_content_length}，触发降级",
            )
            return None

        if should_reject_page(normalized, stage="纯文本检测"):
            return None

        return normalized
```

## 6.2 页面分类器

```python
from enum import Enum


class PdfPageType(str, Enum):
    TEXT = "text"
    MIXED = "mixed"
    SCANNED = "scanned"
    EMPTY = "empty"


class PyMuPdfPageClassifier:
    def __init__(
        self,
        *,
        min_text_chars: int = 80,
        image_area_ratio: float = 0.6,
    ):
        self.min_text_chars = min_text_chars
        self.image_area_ratio = image_area_ratio

    def classify(self, page) -> PdfPageType:
        text = page.get_text("text").strip()
        images = page.get_images(full=True)

        if len(text) >= self.min_text_chars and not images:
            return PdfPageType.TEXT

        if len(text) >= self.min_text_chars and images:
            return PdfPageType.MIXED

        if images and self._has_large_image_area(page):
            return PdfPageType.SCANNED

        if text:
            return PdfPageType.MIXED

        return PdfPageType.EMPTY

    def _has_large_image_area(self, page) -> bool:
        page_area = float(page.rect.width * page.rect.height)
        if page_area <= 0:
            return False

        image_area = 0.0

        for image in page.get_images(full=True):
            xref = image[0]

            try:
                rects = page.get_image_rects(xref)
            except Exception:
                continue

            for rect in rects:
                image_area += float(rect.width * rect.height)

        return image_area / page_area >= self.image_area_ratio
```

```python
class PyMuPdfPageRenderer:
    def __init__(self, *, dpi: int = 200):
        self.dpi = dpi

    def render_page(self, page) -> np.ndarray:
        fitz = load_pymupdf()

        zoom = self.dpi / 72
        matrix = fitz.Matrix(zoom, zoom)

        pix = page.get_pixmap(matrix=matrix, alpha=False)
```

## `camelot_table_extractor.py`

```python
from pathlib import Path

from chat.application.document_parse.schema import ParsedTable


class CamelotTableExtractor:
    def extract_page_tables(
        self,
        *,
        pdf_path: Path,
        page_index: int,
    ) -> list[ParsedTable]:
        page_number = page_index + 1

        for flavor in ("lattice", "stream"):
            tables = self._read_pdf_page(
                pdf_path=pdf_path,
                page_number=page_number,
                flavor=flavor,
            )

            if not tables or tables.n == 0:
                continue

            result: list[ParsedTable] = []

            for table_index, table in enumerate(tables):
                rows = table.df.astype(str).values.tolist()

                if not rows:
                    continue

                result.append(
                    ParsedTable(
                        table_id=f"pdf_p{page_index}_camelot_{flavor}_{table_index}",
                        source="camelot",
                        rows=rows,
                        page_index=page_index,
                        metadata={
                            "flavor": flavor,
                            "parsing_report": getattr(table, "parsing_report", None),
                        },
                    )
                )

            return result

        return []

    def _read_pdf_page(self, *, pdf_path: Path, page_number: int, flavor: str):
        import camelot

        try:
            return camelot.read_pdf(
                str(pdf_path),
                pages=str(page_number),
                flavor=flavor,
            )
        except Exception:
            return None
```

# 8. PaddleOCR 文本 OCR 适配

当前 `LocalOcrProcessor` 价值在于它已经有 lazy persistent worker、worker 协议、超时、worker reset、stderr drain 等能力；但它当前的 `extract_from_url()` 会自己下载 URL，这条路径不能被新 `document_parse` 使用。

新 `document_parse` 只能把扫描 PDF 页渲染成图片文件后交给 OCR worker。当前 worker 支持传入本地 `input_path`，并返回 JSON 文本结果；`paddle_ocr_worker` 内部会初始化 `PaddleOCR` 并从结果中收集文本。

## `local_ocr_image_adapter.py`

```python
from pathlib import Path
from typing import Any


class LocalOcrImageAdapter:
    """
    只用于扫描 PDF 页的本地图片 OCR。

    不处理 URL。
    不处理独立图片文件。
    不做下载。
    """

    def __init__(self, *, local_ocr_processor: Any):
        self.local_ocr_processor = local_ocr_processor

    async def recognize_image_path(self, image_path: Path) -> str:
        if not image_path.is_file():
            raise FileNotFoundError(image_path)

        if hasattr(self.local_ocr_processor, "recognize_image_path"):
            result = self.local_ocr_processor.recognize_image_path(image_path)
            return await self._maybe_await(result)

        if hasattr(self.local_ocr_processor, "_recognize_image_text"):
            result = self.local_ocr_processor._recognize_image_text(image_path)
            return await self._maybe_await(result)

        raise AttributeError(
            "LocalOcrProcessor must expose recognize_image_path(path) "
            "or existing _recognize_image_text(path)."
        )

    async def _maybe_await(self, value):
        if hasattr(value, "__await__"):
            return await value
        return value
```

# 9. PP-Structure 表格抽取

PP-Structure 是 PaddleOCR 的文档结构解析系统，面向 layout analysis、table recognition 等文档理解任务。([PaddlePaddle][3])

架构文档已明确：PP-Structure 需要单独压测，不能默认等同于 PaddleOCR 文本 OCR 的稳定性；失败时必须输出 PaddleOCR 纯文本并标记 `table_parse_failed`。

## `pp_structure_table_extractor.py`

```python
from bs4 import BeautifulSoup

from chat.application.document_parse.schema import ParsedTable


class PPStructureTableExtractor:
    def __init__(self, *, lang: str = "ch"):
        self.lang = lang
        self._engine = None

    @property
    def engine(self):
        if self._engine is None:
            from paddleocr import PPStructure

            self._engine = PPStructure(
                show_log=False,
                lang=self.lang,
            )

        return self._engine

    def extract_page_tables(
        self,
        *,
        image_path,
        page_index: int,
    ) -> tuple[list[ParsedTable], list[str]]:
        warnings: list[str] = []

        try:
            result = self.engine(str(image_path))
        except TypeError:
            try:
                result = self.engine(image_path)
            except Exception as exc:
                return [], [f"table_parse_failed: {type(exc).__name__}: {exc}"]
        except Exception as exc:
            return [], [f"table_parse_failed: {type(exc).__name__}: {exc}"]

        tables: list[ParsedTable] = []

        for item in result or []:
            if item.get("type") != "table":
                continue

            html = item.get("res", {}).get("html", "") or ""
            rows = self._html_table_to_rows(html)

            if not rows:
                continue

            tables.append(
                ParsedTable(
                    table_id=f"pdf_p{page_index}_ppstructure_{len(tables)}",
                    source="pp_structure",
                    rows=rows,
                    page_index=page_index,
                    metadata={
                        "bbox": item.get("bbox"),
                        "html": html,
                    },
                )
            )

        return tables, warnings

    def _html_table_to_rows(self, html: str) -> list[list[str]]:
        soup = BeautifulSoup(html, "html.parser")

        rows: list[list[str]] = []
        for tr in soup.find_all("tr"):
            cells = [
                cell.get_text(strip=True)
                for cell in tr.find_all(["th", "td"])
            ]
            if cells:
                rows.append(cells)

        return rows
```

# 10. PDF Parser

## `pdf_parser.py`

```python
import tempfile
from pathlib import Path

from chat.application.document_parse.schema import DocumentParseResult, ParsedPage
from chat.application.document_parse.pdf.pymupdf_page_classifier import PdfPageType


class PdfParser:
    def __init__(
        self,
        *,
        classifier,
        text_extractor,
        page_renderer,
        camelot_table_extractor,
        ocr_adapter,
        pp_structure_table_extractor,
    ):
        self.classifier = classifier
        self.text_extractor = text_extractor
        self.page_renderer = page_renderer
        self.camelot_table_extractor = camelot_table_extractor
        self.ocr_adapter = ocr_adapter
        self.pp_structure_table_extractor = pp_structure_table_extractor

    async def parse(self, path: Path) -> DocumentParseResult:
        fitz = load_pymupdf()
        doc = fitz.open(path)

        pages: list[ParsedPage] = []
        all_tables = []
        text_parts: list[str] = []
        warnings: list[str] = []

        try:
            with tempfile.TemporaryDirectory(prefix="document-parse-pdf-") as tmp_dir:
                tmp_dir_path = Path(tmp_dir)

                for page_index in range(len(doc)):
                    page = doc[page_index]

                    try:
                        parsed_page = await self._parse_page(
                            pdf_path=path,
                            page=page,
                            page_index=page_index,
                            tmp_dir=tmp_dir_path,
                        )
                    except Exception as exc:
                        warnings.append(
                            f"page_parse_failed: page={page_index + 1}: "
                            f"{type(exc).__name__}: {exc}"
                        )
                        continue

                    pages.append(parsed_page)
                    all_tables.extend(parsed_page.tables)

                    if parsed_page.text:
                        text_parts.append(parsed_page.text)

            return DocumentParseResult(
                text="\n\n".join(text_parts).strip(),
                source=str(path),
                file_type="pdf",
                pages=pages,
                tables=all_tables,
                metadata={
                    "parser": "PdfParser",
                    "pdf_backend": "pymupdf",
                    "table_backends": ["camelot", "pp_structure"],
                    "ocr_backend": "paddleocr",
                },
                warnings=warnings,
            )

        finally:
            doc.close()

    async def _parse_page(
        self,
        *,
        pdf_path: Path,
        page,
        page_index: int,
        tmp_dir: Path,
    ) -> ParsedPage:
        page_type = self.classifier.classify(page)

        if page_type in {PdfPageType.TEXT, PdfPageType.MIXED}:
            text = self.text_extractor.extract_text(page)

            tables = self.camelot_table_extractor.extract_page_tables(
                pdf_path=pdf_path,
                page_index=page_index,
            )

            return ParsedPage(
                page_index=page_index,
                text=text,
                page_type=page_type.value,
                tables=tables,
                metadata={
                    "ocr_used": False,
                    "table_backend": "camelot",
                },
            )

        if page_type == PdfPageType.SCANNED:
            image_path = tmp_dir / f"page_{page_index + 1}.png"
            self.page_renderer.render_page_to_path(page, image_path)

            text = await self.ocr_adapter.recognize_image_path(image_path)

            tables, table_warnings = self.pp_structure_table_extractor.extract_page_tables(
                image_path=image_path,
                page_index=page_index,
            )

            return ParsedPage(
                page_index=page_index,
                text=text,
                page_type=page_type.value,
                tables=tables,
                metadata={
                    "ocr_used": True,
                    "table_backend": "pp_structure",
                    "warnings": table_warnings,
                },
            )

        return ParsedPage(
            page_index=page_index,
            text="",
            page_type=page_type.value,
            tables=[],
            metadata={
                "ocr_used": False,
            },
        )
```

### `PyMuPdfPageRenderer` 需要提供落盘方法

```python
class PyMuPdfPageRenderer:
    def __init__(self, *, dpi: int = 200, max_image_pixels: int = 24_000_000):
        self.dpi = dpi
        self.max_image_pixels = max_image_pixels

    def render_page_to_path(self, page, output_path):
        fitz = load_pymupdf()

        zoom = self.dpi / 72
        matrix = fitz.Matrix(zoom, zoom)

        estimated_width = int(page.rect.width * zoom)
        estimated_height = int(page.rect.height * zoom)
        estimated_pixels = estimated_width * estimated_height

        if estimated_pixels > self.max_image_pixels:
            raise ValueError(
                f"page image too large: {estimated_pixels} > {self.max_image_pixels}"
            )

        pix = page.get_pixmap(matrix=matrix, alpha=False)

        actual_pixels = pix.width * pix.height
        if actual_pixels > self.max_image_pixels:
            raise ValueError(
                f"page image too large: {actual_pixels} > {self.max_image_pixels}"
            )

        pix.save(str(output_path))
        return output_path
```

## 11.3 `lightweight_office_fallback.py`

```python
from pathlib import Path

from chat.application.document_parse.schema import DocumentParseResult, ParsedPage, ParsedTable


class LightweightOfficeFallback:
    def parse(self, path: Path, *, file_type: str) -> DocumentParseResult:
        if file_type == "docx":
            return self._parse_docx(path)

        if file_type == "pptx":
            return self._parse_pptx(path)

        raise ValueError(f"Unsupported office fallback type: {file_type}")

    def _parse_docx(self, path: Path) -> DocumentParseResult:
        import docx

        document = docx.Document(path)

        parts: list[str] = []
        tables: list[ParsedTable] = []

        for paragraph in document.paragraphs:
            text = paragraph.text.strip()
            if text:
                parts.append(text)

        for table_index, table in enumerate(document.tables):
            rows = []
            for row in table.rows:
                cells = [cell.text.strip() for cell in row.cells]
                if any(cells):
                    rows.append(cells)

            if rows:
                tables.append(
                    ParsedTable(
                        table_id=f"docx_table_{table_index}",
                        source="python_docx",
                        rows=rows,
                        metadata={},
                    )
                )
                parts.append(f"\n[Table {table_index}]")
                parts.extend([" | ".join(row) for row in rows])

        text = "\n\n".join(parts).strip()

        page = ParsedPage(
            page_index=0,
            text=text,
            page_type="document",
            tables=tables,
            metadata={"parser": "python_docx"},
        )

        return DocumentParseResult(
            text=text,
            source=str(path),
            file_type="docx",
            pages=[page],
            tables=tables,
            metadata={"parser": "python_docx"},
            warnings=[],
        )

    def _parse_pptx(self, path: Path) -> DocumentParseResult:
        from pptx import Presentation

        presentation = Presentation(path)

        pages: list[ParsedPage] = []
        all_tables: list[ParsedTable] = []
        text_parts: list[str] = []

        for slide_index, slide in enumerate(presentation.slides):
            slide_parts = [f"## Slide {slide_index + 1}"]
            slide_tables: list[ParsedTable] = []

            for shape in slide.shapes:
                if getattr(shape, "has_text_frame", False):
                    text = shape.text.strip()
                    if text:
                        slide_parts.append(text)

                if getattr(shape, "has_table", False):
                    rows = []
                    for row in shape.table.rows:
                        cells = [cell.text.strip() for cell in row.cells]
                        if any(cells):
                            rows.append(cells)

                    if rows:
                        table = ParsedTable(
                            table_id=f"pptx_slide_{slide_index}_table_{len(slide_tables)}",
                            source="python_pptx",
                            rows=rows,
                            page_index=slide_index,
                            metadata={},
                        )
                        slide_tables.append(table)
                        all_tables.append(table)
                        slide_parts.append("[Table]")
                        slide_parts.extend([" | ".join(row) for row in rows])

            slide_text = "\n".join(slide_parts).strip()
            text_parts.append(slide_text)

            pages.append(
                ParsedPage(
                    page_index=slide_index,
                    text=slide_text,
                    page_type="slide",
                    tables=slide_tables,
                    metadata={"parser": "python_pptx"},
                )
            )

        text = "\n\n".join(text_parts).strip()

        return DocumentParseResult(
            text=text,
            source=str(path),
            file_type="pptx",
            pages=pages,
            tables=all_tables,
            metadata={"parser": "python_pptx"},
            warnings=[],
        )
```

# 14. DocumentParseService

```python
from pathlib import Path

from chat.application.document_parse.errors import UnsupportedDocumentTypeError


class DocumentParseService:
    def __init__(
        self,
        *,
        pdf_parser,
        office_parser,
        epub_parser,
        spreadsheet_parser,
    ):
        self.pdf_parser = pdf_parser
        self.office_parser = office_parser
        self.epub_parser = epub_parser
        self.spreadsheet_parser = spreadsheet_parser

    async def parse_path(self, path: Path):
        suffix = path.suffix.lower()

        if suffix == ".pdf":
            return await self.pdf_parser.parse(path)

        if suffix in {".docx", ".pptx"}:
            return self.office_parser.parse(
                path,
                file_type=suffix.lstrip("."),
            )

        if suffix == ".epub":
            return self.epub_parser.parse(path)

        if suffix in {".xlsx", ".xls", ".ods"}:
            return self.spreadsheet_parser.parse(path)

        raise UnsupportedDocumentTypeError(f"Unsupported document suffix: {suffix}")

    async def parse_bytes(self, data: bytes, *, filename: str):
        import tempfile

        suffix = Path(filename).suffix.lower()
        if not suffix:
            raise UnsupportedDocumentTypeError("Missing file suffix")

        with tempfile.NamedTemporaryFile(suffix=suffix, delete=False) as tmp:
            tmp.write(data)
            tmp_path = Path(tmp.name)

        try:
            return await self.parse_path(tmp_path)
        finally:
            tmp_path.unlink(missing_ok=True)
```

# 15. DocumentParseTool

工具 schema 按仓库现有工具注册方式接入。此处只给核心逻辑。

```python
from pathlib import Path
from typing import Any, Dict, Optional

from chat.application.tool_content_store import cache_and_window, format_windowed_content
from chat.core.config.app_settings import settings
from chat.domain.interfaces.tool import BaseTool


TOOL_SCHEMA = {
    "type": "object",
    "properties": {
        "file_ref": {
            "type": "string",
            "description": "Reference to a local or cached binary document file.",
        },
    },
    "required": ["file_ref"],
}


class DocumentParseTool(BaseTool):
    def __init__(self, *, parse_service, file_resolver):
        self.parse_service = parse_service
        self.file_resolver = file_resolver

    @property
    def name(self) -> str:
        return "document_parse"

    @property
    def description(self) -> str:
        return (
            "Parses binary documents into Markdown text and structured tables. "
            "Supports PDF, DOCX, PPTX, EPUB, XLSX, XLS, and ODS. "
            "Does not handle HTML, TXT, MD, CSV, JSON, XML, images, audio, or video."
        )

    @property
    def parameters_schema(self) -> Dict[str, Any]:
        return TOOL_SCHEMA

    async def execute(self, context: Dict[str, Any], **kwargs) -> str:
        session_id: Optional[str] = context.get("session_id")
        if not session_id:
            return "[Tool Error] Missing session_id in execution context."

        file_ref = str(kwargs.get("file_ref", "")).strip()
        if not file_ref:
            return "[Tool Error] Missing required file_ref parameter"

        resolved = await self._resolve_file(file_ref)
        path = Path(resolved.local_path)

        result = await self.parse_service.parse_path(path)

        metadata = {
            **result.metadata,
            "tool": self.name,
            "file_type": result.file_type,
            "source": result.source,
            "page_count": len(result.pages),
            "table_count": len(result.tables),
            "warnings": result.warnings,
        }

        window = cache_and_window(
            session_id=session_id,
            tool_name=self.name,
            source=result.source,
            text=result.text,
            content_type="text/markdown",
            metadata=metadata,
            offset=0,
            limit=settings.TOOL_RESULT_MAX_CHARS,
        )

        return format_windowed_content(window)

    async def _resolve_file(self, file_ref: str):
        result = self.file_resolver.resolve(file_ref)
        if hasattr(result, "__await__"):
            result = await result
        return result
```

说明：

```text
file_resolver 使用仓库已有文件引用 / 二进制缓存 / 本地文件解析机制。
不要在 document_parse 内实现 URL 下载。
```
