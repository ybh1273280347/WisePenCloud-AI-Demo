# 文档一：通用工具长内容缓存与分段读取

## 0. 背景

当前 `web_fetch` 的底层抓取和解析可能已经拿到了完整内容，但 `WebFetchTool` 返回给模型前会按 `settings.TOOL_RESULT_MAX_CHARS` 截断。这样会导致长论文、长公告、长网页只返回开头部分，模型并不知道自己没有读完整。

社区里处理 agent 工具输出的常见做法是：**工具输出必须有大小上限，但截断必须显式，并提供 offset/limit 等可执行续读方式**。相关 agent harness 实践中也明确提到，大输出应通过 offset/limit 增量处理，而不是让模型在不知道完整长度的情况下基于截断内容回答。([arXiv][1])

本次目标不是只修 `web_fetch`，而是给以后所有工具预留统一扩展点：

```text
完整内容进入 ToolContentStore
工具返回受控窗口
模型通过 content_id + offset 继续读取
```

---

## 1. 本次目标

实现：

```text
1. 新增 ToolContentStore，保存完整工具输出
2. 新增 tool_content_read 工具，用 content_id + offset + limit 续读
3. web_fetch 接入 ToolContentStore
4. web_fetch 不再只返回简单截断，而是返回 metadata + 当前窗口内容
5. 保持单次工具输出受 settings.TOOL_RESULT_MAX_CHARS 控制
6. 按 session_id 隔离缓存内容，避免跨会话读取
```

不做：

```text
1. 不上 Redis / SQLite / 文件缓存
2. 不做向量库 / RAG
3. 不做 search_cached
4. 不取消 TOOL_RESULT_MAX_CHARS
5. 不把完整论文一次性塞给模型
```

### 3.2 完整代码

```python
import time
import uuid
from collections import OrderedDict
from dataclasses import dataclass
from typing import Any, Dict, Optional


_DEFAULT_TTL_SECONDS = 30 * 60
_DEFAULT_MAX_ITEMS = 128


@dataclass
class StoredToolContent:
    content_id: str
    session_id: str
    tool_name: str
    source: str
    content_type: str
    text: str
    created_at: float
    metadata: Dict[str, Any]


@dataclass
class WindowedContent:
    content_id: str
    tool_name: str
    source: str
    content_type: str
    original_length: int
    offset: int
    returned_length: int
    truncated: bool
    next_offset: Optional[int]
    text: str
    error: Optional[str] = None


class ToolContentStore:
    """进程内工具内容缓存。

    第一版只做内存 TTL + LRU：
    - 不持久化
    - 不跨进程
    - 按 session_id 隔离读取
    """

    def __init__(
        self,
        ttl_seconds: int = _DEFAULT_TTL_SECONDS,
        max_items: int = _DEFAULT_MAX_ITEMS,
    ):
        self._ttl_seconds = ttl_seconds
        self._max_items = max_items
        self._items: OrderedDict[str, StoredToolContent] = OrderedDict()

    def put(
        self,
        *,
        session_id: str,
        tool_name: str,
        source: str,
        text: str,
        content_type: str = "text/markdown",
        metadata: Optional[Dict[str, Any]] = None,
    ) -> str:
        self._evict_expired()

        content_id = f"{tool_name}:{uuid.uuid4().hex[:16]}"

        self._items[content_id] = StoredToolContent(
            content_id=content_id,
            session_id=session_id,
            tool_name=tool_name,
            source=source,
            content_type=content_type,
            text=text,
            created_at=time.monotonic(),
            metadata=metadata or {},
        )
        self._items.move_to_end(content_id)
        self._evict_overflow()

        return content_id

    def get(self, *, content_id: str, session_id: str) -> Optional[StoredToolContent]:
        item = self._items.get(content_id)

        if item is None:
            return None

        if item.session_id != session_id:
            return None

        if time.monotonic() - item.created_at > self._ttl_seconds:
            self._items.pop(content_id, None)
            return None

        self._items.move_to_end(content_id)
        return item

    def read_window(
        self,
        *,
        content_id: str,
        session_id: str,
        offset: int,
        limit: int,
    ) -> Optional[WindowedContent]:
        item = self.get(content_id=content_id, session_id=session_id)

        if item is None:
            return None

        text = item.text.strip()
        original_length = len(text)

        offset = max(0, offset)
        limit = max(1, limit)

        if offset >= original_length:
            return WindowedContent(
                content_id=content_id,
                tool_name=item.tool_name,
                source=item.source,
                content_type=item.content_type,
                original_length=original_length,
                offset=offset,
                returned_length=0,
                truncated=False,
                next_offset=None,
                text="",
                error="offset_out_of_range",
            )

        start = offset
        end = min(start + limit, original_length)
        chunk = text[start:end]

        if end < original_length:
            paragraph_boundary = chunk.rfind("\n\n")
            if paragraph_boundary >= int(len(chunk) * 0.6):
                chunk = chunk[:paragraph_boundary]
                end = start + paragraph_boundary

        chunk = chunk.strip()
        truncated = end < original_length

        return WindowedContent(
            content_id=content_id,
            tool_name=item.tool_name,
            source=item.source,
            content_type=item.content_type,
            original_length=original_length,
            offset=offset,
            returned_length=len(chunk),
            truncated=truncated,
            next_offset=end if truncated else None,
            text=chunk,
        )

    def _evict_expired(self) -> None:
        now = time.monotonic()
        expired_keys = [
            key
            for key, item in self._items.items()
            if now - item.created_at > self._ttl_seconds
        ]

        for key in expired_keys:
            self._items.pop(key, None)

    def _evict_overflow(self) -> None:
        while len(self._items) > self._max_items:
            self._items.popitem(last=False)


tool_content_store = ToolContentStore()
```

from typing import Any, Dict, Optional

from chat.core.config.app_settings import settings
from chat.domain.interfaces.tool import BaseTool
from chat.application.tool_content_store import tool_content_store, WindowedContent


class ToolContentReadTool(BaseTool):
    """读取已缓存的长工具内容"""

    @property
    def name(self) -> str:
        return "tool_content_read"

    @property
    def description(self) -> str:
        return (
            "Read a window of previously cached tool content by content_id. "
            "Use this when another tool returns ToolContent Metadata with truncated=true and next_offset. "
            "For full-document reading or summary, continue calling this tool with offset=next_offset until truncated=false."
        )

    @property
    def parameters_schema(self) -> Dict[str, Any]:
        return {
            "type": "object",
            "properties": {
                "content_id": {
                    "type": "string",
                    "description": "The content_id returned in ToolContent Metadata.",
                },
                "offset": {
                    "type": "integer",
                    "description": "Character offset to continue reading from.",
                    "default": 0,
                },
                "limit": {
                    "type": "integer",
                    "description": (
                        "Maximum number of characters to return. "
                        "Defaults to TOOL_RESULT_MAX_CHARS and is capped by TOOL_RESULT_MAX_CHARS."
                    ),
                },
            },
            "required": ["content_id"],
        }

    async def execute(self, context: Dict[str, Any], **kwargs) -> str:
        session_id: Optional[str] = context.get("session_id")
        if not session_id:
            return "[Tool Error] Missing session_id in execution context."

        content_id = kwargs.get("content_id", "")
        if not content_id:
            return "[Tool Error] Missing required content_id parameter"

        offset = _parse_int(kwargs.get("offset"), 0)
        limit = _parse_int(kwargs.get("limit"), settings.TOOL_RESULT_MAX_CHARS)
        limit = min(max(1, limit), settings.TOOL_RESULT_MAX_CHARS)

        window = tool_content_store.read_window(
            content_id=content_id,
            session_id=session_id,
            offset=offset,
            limit=limit,
        )

        if window is None:
            return "[Tool Result] Cached tool content not found or expired."

        return _format_windowed_content(window)


def _parse_int(value: Any, default: int) -> int:
    try:
        return int(value)
    except (TypeError, ValueError):
        return default


def _format_windowed_content(window: WindowedContent) -> str:
    next_offset = "" if window.next_offset is None else str(window.next_offset)

    metadata = (
        "[ToolContent Metadata]\n"
        f"content_id: {window.content_id}\n"
        f"tool_name: {window.tool_name}\n"
        f"source: {window.source}\n"
        f"content_type: {window.content_type}\n"
        f"original_length: {window.original_length}\n"
        f"offset: {window.offset}\n"
        f"returned_length: {window.returned_length}\n"
        f"truncated: {str(window.truncated).lower()}\n"
        f"next_offset: {next_offset}\n"
    )

    if window.error:
        metadata += f"error: {window.error}\n"

    return metadata + "\n[Content]\n" + window.text
```

async def execute(self, context: Dict[str, Any], **kwargs) -> str:
    """执行网页浏览并返回 Markdown 内容或错误消息"""
    session_id: Optional[str] = context.get("session_id")
    if not session_id:
        return "[Tool Error] Missing session_id in execution context."

    url: str = kwargs.get("url", "")
    force_browser: bool = kwargs.get("force_browser", False)

    if not url:
        return "[Tool Error] Missing required url parameter"

    url = url.strip()
    if not _is_valid_http_url(url):
        return "[Tool Error] Invalid url parameter. URL must start with http:// or https://."

    try:
        md_result = await self._fetcher.fetch(url, force_browser=force_browser)
    except Exception as e:
        log_fail("网页抓取工具", e, session_id=session_id, url=url, force_browser=force_browser)
        return "[Tool Error] Unexpected error while fetching web page content."

    if md_result is None:
        return "[Tool Result] Failed to fetch web page content (all fetch methods exhausted)"

    md_result = md_result.strip()

    if not md_result:
        return "[Tool Result] Failed to fetch web page content (empty content returned)"

    offset = _parse_int(kwargs.get("offset"), 0)
    limit = _parse_int(kwargs.get("limit"), settings.TOOL_RESULT_MAX_CHARS)
    limit = min(max(1, limit), settings.TOOL_RESULT_MAX_CHARS)

    content_id = tool_content_store.put(
        session_id=session_id,
        tool_name=self.name,
        source=url,
        text=md_result,
        content_type="text/markdown",
        metadata={
            "force_browser": force_browser,
        },
    )

    window = tool_content_store.read_window(
        content_id=content_id,
        session_id=session_id,
        offset=offset,
        limit=limit,
    )

    if window is None:
        return "[Tool Result] Failed to read cached web fetch content."

    return _format_windowed_content(window)
```

## 7. 第一版的重要取舍

### 7.1 为什么 content_id 不用 URL hash

使用随机 `uuid`：

```text
避免 content_id 可猜
避免跨 session 猜测读取
允许同一 URL 多次抓取生成不同版本
```

### 7.2 为什么按 session_id 隔离

后续其他工具可能缓存登录后内容、本地浏览器内容、邮件内容。
必须从第一版就防止跨 session 读取。

### 7.3 为什么 web_fetch 每次都会 put

第一版简单可靠。
后续可以优化为：

```text
同一 session + URL + force_browser 命中已有 content_id
```

当前先不做，避免复杂。

# 文档二：使用 Docling 升级文档解析质量

## 0. 背景

当前文档解析已经能读 PDF/DOCX/XLSX/PPTX，但 PDF 出现了明显质量问题：

```text
段落粘连
句子中间没有空格
长文本变成超长 token
多栏阅读顺序不稳定
表格结构不够清楚
```

这个问题不是简单文本 normalize 可以彻底解决的。PDF 本质是版面格式，不是语义文档格式，需要更强的 layout / reading order / table structure 解析能力。

Docling 官方支持 PDF、DOCX、XLSX、PPTX、HTML 等输入，并支持导出 Markdown、JSON、Text、HTML、DocTags 等格式；它还强调 advanced PDF understanding，包括 page layout、reading order、table structure、code、formulas、image classification 等能力。([DocLing Project][2])

Docling 的 `DocumentConverter` 是官方主入口，用于把 PDF、DOCX、PPTX、HTML、Markdown 等输入转换成 DoclingDocument，并可导出 Markdown。([DocLing Project][3])

---

## 1. 最终决策

第一版采用：

```text
Docling 作为 PDF / PPTX / DOCX 主解析引擎
XLSX 继续使用 native openpyxl
Legacy parser 保留为兼容 fallback
不引入 Unstructured
不做复杂 parser 编排器
```

原因：

```text
1. 主要痛点是 PDF/PPTX/DOCX 的结构化阅读质量
2. Docling 正好擅长文档转 Markdown / 结构化输出
3. XLSX 用 openpyxl 输出 Sheet + TSV 更可控
4. Unstructured 更偏 partition / RAG ingest，当前阶段不引入
5. 文档解析通常不是“失败降级”问题，而是“选一个强主引擎”
```

## 5. `document_parser.py` 完整实现方案

下面代码可作为第一版主体。

````python
from io import BytesIO
from typing import Callable, Dict, List, Optional
from zipfile import BadZipFile, ZipFile
import tempfile

import pdfplumber
from docx import Document as DocxDocument
from openpyxl import load_workbook
from pptx import Presentation

from chat.core.config.app_settings import settings
from common.logger import log_ok, log_fail


_OLE_MAGIC = b"\xd0\xcf\x11\xe0\xa1\xb1\x1a\xe1"

_MAX_ZIP_UNCOMPRESSED_SIZE = 200 * 1024 * 1024
_DOCLING_DOC_TYPES = {"pdf", "docx", "pptx"}


def _normalize_text(text: str) -> str:
    lines = [
        line.rstrip()
        for line in text.replace("\r\n", "\n").replace("\r", "\n").split("\n")
    ]
    result = "\n".join(lines).strip()

    while "\n\n\n" in result:
        result = result.replace("\n\n\n", "\n\n")

    return result


class DocumentParser:
    """文档解析器。

    第一版设计：
    - PDF/DOCX/PPTX 优先使用 Docling
    - XLSX 继续使用 openpyxl
    - legacy parser 只作为兼容 fallback
    """

    def __init__(
        self,
        min_content_length: int = 50,
        max_document_size: int = 50 * 1024 * 1024,
    ):
        self._min_content_length = min_content_length
        self._max_document_size = max_document_size
        self._docling_converter = None

        self._legacy_parsers: Dict[str, Callable[[bytes], Optional[str]]] = {
            "pdf": self._parse_pdf_legacy,
            "docx": self._parse_docx_legacy,
            "xlsx": self._parse_xlsx_legacy,
            "pptx": self._parse_pptx_legacy,
        }

    def parse(self, data: bytes) -> Optional[str]:
        if len(data) > self._max_document_size:
            log_fail("文档解析", f"文件过大({len(data)}字节)，上限{self._max_document_size}字节")
            return None

        doc_type = self._detect_doc_type(data)
        if doc_type is None:
            return None

        engine = getattr(settings, "DOCUMENT_PARSER_ENGINE", "docling")
        enable_legacy_fallback = getattr(settings, "DOCUMENT_PARSER_ENABLE_LEGACY_FALLBACK", True)

        if engine == "docling" and doc_type in _DOCLING_DOC_TYPES:
            text = self._parse_with_docling(data, doc_type)

            if text:
                return self._finalize(
                    text,
                    doc_type=doc_type,
                    engine="docling",
                )

            if not enable_legacy_fallback:
                return None

        text = self._parse_with_legacy(data, doc_type)
        if text:
            return self._finalize(
                text,
                doc_type=doc_type,
                engine="legacy",
            )

        return None

    def _detect_doc_type(self, data: bytes) -> Optional[str]:
        if data[:5] == b"%PDF-":
            return "pdf"

        if data[:8] == _OLE_MAGIC:
            log_fail("文档解析", "OLE 复合文档(旧版 .doc/.xls/.ppt)，暂不支持")
            return None

        try:
            with ZipFile(BytesIO(data)) as zf:
                infos = zf.infolist()
                uncompressed_size = sum(info.file_size for info in infos)

                if uncompressed_size > _MAX_ZIP_UNCOMPRESSED_SIZE:
                    log_fail(
                        "文档解析",
                        f"ZIP 解压后体积过大({uncompressed_size}字节)，上限{_MAX_ZIP_UNCOMPRESSED_SIZE}字节",
                    )
                    return None

                names = {info.filename for info in infos}

                matches = [
                    ("word/document.xml", "docx"),
                    ("xl/workbook.xml", "xlsx"),
                    ("ppt/presentation.xml", "pptx"),
                ]

                hits = [doc_type for sig, doc_type in matches if sig in names]

                if len(hits) != 1:
                    if hits:
                        log_fail("文档解析", f"ZIP 内含多种 Office 特征文件({', '.join(hits)})，无法确定类型")
                    return None

                return hits[0]

        except BadZipFile:
            pass

        log_fail("文档解析", "无法识别文档类型")
        return None

    def _parse_with_docling(self, data: bytes, doc_type: str) -> Optional[str]:
        if doc_type not in _DOCLING_DOC_TYPES:
            return None

        try:
            converter = self._get_docling_converter()
            if converter is None:
                return None

            suffix = f".{doc_type}"

            with tempfile.NamedTemporaryFile(suffix=suffix, delete=True) as tmp:
                tmp.write(data)
                tmp.flush()

                result = converter.convert(tmp.name)
                markdown = result.document.export_to_markdown()

            markdown = markdown.strip()
            if markdown:
                log_ok("Docling 文档解析", doc_type=doc_type, length=len(markdown))
                return markdown

            return None

        except Exception as e:
            log_fail("Docling 文档解析", e, doc_type=doc_type)
            return None

    def _get_docling_converter(self):
        if self._docling_converter is not None:
            return self._docling_converter

        try:
            from docling.document_converter import DocumentConverter
        except Exception as e:
            log_fail("Docling 初始化", f"未能导入 Docling: {e}")
            return None

        try:
            self._docling_converter = DocumentConverter()
            return self._docling_converter
        except Exception as e:
            log_fail("Docling 初始化", e)
            return None

    def _parse_with_legacy(self, data: bytes, doc_type: str) -> Optional[str]:
        parser = self._legacy_parsers.get(doc_type)

        if parser is None:
            log_fail("文档解析", f"暂不支持的文档类型: {doc_type}")
            return None

        return parser(data)

    def _finalize(self, text: str, *, doc_type: str, engine: str) -> Optional[str]:
        cleaned = _normalize_text(text)

        if len(cleaned) < self._min_content_length:
            log_fail(
                "文档清洗",
                f"提取文本过短({len(cleaned)}字符)，阈值{self._min_content_length}",
                doc_type=doc_type,
                engine=engine,
            )
            return None

        self._log_quality_warnings(cleaned, doc_type=doc_type, engine=engine)

        log_ok(
            "文档清洗",
            length=len(cleaned),
            doc_type=doc_type,
            engine=engine,
        )
        return cleaned

    def _log_quality_warnings(self, text: str, *, doc_type: str, engine: str) -> None:
        longest_token = max((len(token) for token in text.split()), default=0)

        if longest_token > 120:
            log_fail(
                "文档解析质量告警",
                f"存在超长 token({longest_token})，可能出现文本粘连",
                doc_type=doc_type,
                engine=engine,
            )

        if len(text) > 2000 and text.count("\n") < 5:
            log_fail(
                "文档解析质量告警",
                "长文档换行过少，可能结构塌陷",
                doc_type=doc_type,
                engine=engine,
            )

    def _parse_pdf_legacy(self, data: bytes) -> Optional[str]:
        try:
            text_parts: List[str] = []
            page_count = 0

            with pdfplumber.open(BytesIO(data)) as pdf:
                page_count = len(pdf.pages)

                for page in pdf.pages:
                    page_text = page.extract_text(layout=True) or page.extract_text()
                    if page_text:
                        text_parts.append(page_text)

            result = "\n\n".join(text_parts).strip()

            if result:
                log_ok("PDF legacy 文本提取", pages=page_count, length=len(result))
                return result

            return None

        except Exception as e:
            log_fail("PDF legacy 文本提取", e)
            return None

    def _parse_docx_legacy(self, data: bytes) -> Optional[str]:
        try:
            doc = DocxDocument(BytesIO(data))
            parts: List[str] = []

            for para in doc.paragraphs:
                text = para.text.strip()
                if text:
                    parts.append(text)

            for table in doc.tables:
                for row in table.rows:
                    cells = [cell.text.strip() for cell in row.cells]
                    if any(cells):
                        parts.append("\t".join(cells))

            result = "\n".join(parts).strip()

            if result:
                log_ok("DOCX legacy 文本提取", length=len(result))
                return result

            return None

        except Exception as e:
            log_fail("DOCX legacy 文本提取", e)
            return None

    def _parse_xlsx_legacy(self, data: bytes) -> Optional[str]:
        wb = None

        try:
            wb = load_workbook(BytesIO(data), read_only=True, data_only=True)
            parts: List[str] = []

            for sheet in wb:
                sheet_rows: List[str] = []

                for row in sheet.iter_rows(values_only=True):
                    cells = [str(cell) if cell is not None else "" for cell in row]
                    line = "\t".join(cells).strip()

                    if line:
                        sheet_rows.append(line)

                if sheet_rows:
                    parts.append(f"## Sheet: {sheet.title}")
                    parts.append("")
                    parts.append("```tsv")
                    parts.extend(sheet_rows)
                    parts.append("```")
                    parts.append("")

            result = "\n".join(parts).strip()

            if result:
                log_ok("XLSX legacy 文本提取", length=len(result))
                return result

            return None

        except Exception as e:
            log_fail("XLSX legacy 文本提取", e)
            return None

        finally:
            if wb is not None:
                wb.close()

    def _parse_pptx_legacy(self, data: bytes) -> Optional[str]:
        try:
            prs = Presentation(BytesIO(data))
            parts: List[str] = []

            for index, slide in enumerate(prs.slides, 1):
                slide_parts: List[str] = []

                for shape in slide.shapes:
                    if shape.has_text_frame:
                        text = shape.text.strip()
                        if text:
                            slide_parts.append(text)

                if slide_parts:
                    parts.append(f"## Slide {index}")
                    parts.append("")
                    parts.extend(slide_parts)
                    parts.append("")

            result = "\n".join(parts).strip()

            if result:
                log_ok("PPTX legacy 文本提取", length=len(result))
                return result

            return None

        except Exception as e:
            log_fail("PPTX legacy 文本提取", e)
            return None

## 8. 验收标准

### 8.1 PDF

必须看到日志：

```text
Docling 文档解析 | doc_type=pdf
文档清洗 | engine=docling
```

输出应明显减少：

```text
超长 token
段落粘连
换行缺失
表格完全塌陷
```

### 8.2 PPTX

PPTX 应优先走 Docling：

```text
Docling 文档解析 | doc_type=pptx
```

输出应比 legacy shape 拼接更接近 Markdown 文档。

### 8.3 DOCX

```text
Docling 文档解析 | doc_type=docx
```

标题、段落、列表、表格应尽量保留结构。

---

```text
XLSX legacy 文本提取
```

输出保持：

## Sheet: xxx
